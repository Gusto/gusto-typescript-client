# PayrollDigestResults

A payroll digest batch with processing results.

## Example Usage

```typescript
import { PayrollDigestResults } from "@gusto/embedded-api-v-2026-06-15/models/components/payrolldigestresults.js";

let value: PayrollDigestResults = {
  uuid: "1f903064-3bf6-4f01-9d7a-fe7d1d2da4b6",
  idempotencyKey: "d7e95006-b89e-43b1-b239-cb0317f66a73",
  status: "processing",
  submittedAt: new Date("2024-12-27T20:49:55.622Z"),
};
```
