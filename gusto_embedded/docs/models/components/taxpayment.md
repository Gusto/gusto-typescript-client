# TaxPayment

Representation of a tax payment made by Gusto to a tax agency on behalf of a company

## Example Usage

```typescript
import { TaxPayment } from "@gusto/embedded-api/models/components/taxpayment.js";
import { RFCDate } from "@gusto/embedded-api/types/rfcdate.js";

let value: TaxPayment = {
  uuid: "97cc2c5f-49ed-4618-a63a-27a37a39c76f",
  companyUuid: "<id>",
  agencyName: "<value>",
  jurisdiction: "<value>",
  periodStart: new RFCDate("2024-09-18"),
  periodEnd: new RFCDate("2026-10-17"),
  dueDate: new RFCDate("2024-09-23"),
  amount: "<value>",
  amountPaid: "<value>",
};
```
