# PayrollBatchResults

A payroll cancellation batch with per-payroll results.

## Example Usage

```typescript
import { PayrollBatchResults } from "@gusto/embedded-api-v-2026-02-01/models/components/payrollbatchresults.js";

let value: PayrollBatchResults = {
  uuid: "9b61c2c4-2476-4a40-ac77-a115e66ebfae",
  idempotencyKey: "c161b4ae-82cf-4381-b87e-208542d86832",
  status: "completed",
  submittedAt: new Date("2026-02-20T02:55:03.710Z"),
};
```
