# BulkReportCompany

Results for a single company in a bulk report batch.

## Example Usage

```typescript
import { BulkReportCompany } from "@gusto/embedded-api-v-2026-02-01/models/components/bulkreportcompany.js";

let value: BulkReportCompany = {
  companyUuid: "a689b265-916e-40f2-84b8-e6110671b8a4",
  status: "success",
  reports: [],
};
```
