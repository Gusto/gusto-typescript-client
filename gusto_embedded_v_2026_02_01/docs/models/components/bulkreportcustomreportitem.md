# BulkReportCustomReportItem

A custom report entry in a bulk batch.

## Example Usage

```typescript
import { BulkReportCustomReportItem } from "@gusto/embedded-api-v-2026-02-01/models/components/bulkreportcustomreportitem.js";
import { RFCDate } from "@gusto/embedded-api-v-2026-02-01/types/rfcdate.js";

let value: BulkReportCustomReportItem = {
  companyUuid: "12345678-abcd-ef12-3456-7890abcdef12",
  reportType: "custom_report",
  columns: [
    "gross_earnings",
  ],
  fileType: "csv",
  startDate: new RFCDate("2026-01-01"),
  endDate: new RFCDate("2026-03-31"),
};
```
